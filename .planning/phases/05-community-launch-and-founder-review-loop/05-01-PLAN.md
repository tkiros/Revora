```typescript
export type LaunchMetricName = "query_completed" | "share_intent";

export type LaunchMetricInput = {
  name: LaunchMetricName;
  day: string;
  week: string;
  environment: "production" | "preview" | "development" | "test";
  responseKind?: "result" | "clarify" | "not_food" | "out_of_scope" | "retry";
  risk?: "SAFE" | "MODERATE" | "HIGH";
};

export async function recordLaunchMetric(input: LaunchMetricInput): Promise<
  | { recorded: true; key: string }
  | { recorded: false; reason: "not_production" | "setup_blocked" }
>;
export async function recordQueryCompleted(input: Omit<LaunchMetricInput, "name">): Promise<unknown>;
export async function recordShareIntent(input: Omit<LaunchMetricInput, "name" | "responseKind" | "risk">): Promise<unknown>;
```
